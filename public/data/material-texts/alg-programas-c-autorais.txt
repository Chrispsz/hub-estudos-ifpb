--- reverso.c --- (decomposição de dígitos com / e % — técnica de ouro)
#include <stdio.h>

void main(){
    int num;
    printf("Informe o numero inteiro entre 1000 e 9999: ");
    scanf("%d", &num);
    int milhar = num / 1000;
    int resto = num % 1000;
    int centena = resto / 100;
    resto = resto % 100;
    int dezena = resto / 10;
    int unidade = resto % 10;
    int reverso = (unidade * 1000)+(dezena * 100)+(centena * 10)+milhar;
    printf("O numero reverso de %d eh %d\n",num, reverso);
}
